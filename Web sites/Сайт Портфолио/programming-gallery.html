Змейка на C++ (консольная)

C++

Классическая игра — работает в любой консоли. Без Windows-специфичных функций.

Показать код
#include #include #include #include #include #include using namespace std; const int WIDTH = 20, HEIGHT = 20; int x = WIDTH/2, y = HEIGHT/2; vector> snake; int foodX, foodY, score = 0; bool gameOver = false; char direction = ' '; void Setup() { srand(time(0)); snake.clear(); snake.push_back({x, y}); foodX = rand() % WIDTH; foodY = rand() % HEIGHT; gameOver = false; direction = ' '; } void Draw() { system("clear"); for(int i = 0; i < WIDTH + 2; i++) cout << "#"; cout << endl; for(int i = 0; i < HEIGHT; i++) { for(int j = 0; j < WIDTH; j++) { if(j == 0) cout << "#"; if(i == y && j == x) cout << "O"; else if(i == foodY && j == foodX) cout << "F"; else { bool print = false; for(int k = 0; k < snake.size(); k++) { if(snake[k].first == j && snake[k].second == i) { cout << "o"; print = true; } } if(!print) cout << " "; } if(j == WIDTH - 1) cout << "#"; } cout << endl; } for(int i = 0; i < WIDTH + 2; i++) cout << "#"; cout << endl; cout << "Score: " << score << endl; } void Input() { // В онлайн-компиляторах не работает kbhit() — используем автопилот } bool kbhit() { return false; } void Logic() { int prevX = snake[0].first; int prevY = snake[0].second; int prev2X, prev2Y; snake[0].first = x; snake[0].second = y; for(int i = 1; i < snake.size(); i++) { prev2X = snake[i].first; prev2Y = snake[i].second; snake[i].first = prevX; snake[i].second = prevY; prevX = prev2X; prevY = prev2Y; } switch(direction) { case 'L': x--; break; case 'R': x++; break; case 'U': y--; break; case 'D': y++; break; } if(x >= WIDTH || x < 0 || y >= HEIGHT || y < 0) gameOver = true; for(int i = 1; i < snake.size(); i++) { if(snake[i].first == x && snake[i].second == y) gameOver = true; } if(x == foodX && y == foodY) { score += 10; foodX = rand() % WIDTH; foodY = rand() % HEIGHT; snake.push_back({prevX, prevY}); } } int main() { Setup(); while(!gameOver) { Draw(); direction = 'R'; // Автопилот для онлайн-компиляторов Logic(); this_thread::sleep_for(chrono::milliseconds(300)); } cout << "Game Over! Score: " << score << endl; return 0; }

Как запустить: Скопируйте код → зайдите на OnlineGDB (C++) → вставьте → нажмите "Run".
Игра запустится в автопилоте — для интерактивности используйте локальный компилятор.

Крестики-нолики на Java (консоль)

Java

Игра в консоли — ввод ходов через цифры 1-9. Никаких Swing!

Показать код
import java.util.Scanner; class Main { private static char[] board = new char[9]; private static char currentPlayer = 'X'; private static Scanner scanner = new Scanner(System.in); public static void main(String[] args) { for(int i = 0; i < 9; i++) board[i] = ' '; while(true) { printBoard(); if(checkWin()) { System.out.println("Победил: " + currentPlayer); break; } if(isBoardFull()) { System.out.println("Ничья!"); break; } System.out.print("Игрок " + currentPlayer + ", введите позицию (1-9): "); int pos = scanner.nextInt() - 1; if(pos >= 0 && pos < 9 && board[pos] == ' ') { board[pos] = currentPlayer; currentPlayer = (currentPlayer == 'X') ? 'O' : 'X'; } else { System.out.println("Неверный ход!"); } } scanner.close(); } private static void printBoard() { for(int i = 0; i < 9; i++) { System.out.print(" " + board[i]); if(i % 3 == 2) System.out.println(); else System.out.print(" |"); } System.out.println(); } private static boolean checkWin() { int[][] winPatterns = { {0,1,2}, {3,4,5}, {6,7,8}, {0,3,6}, {1,4,7}, {2,5,8}, {0,4,8}, {2,4,6} }; for(int[] pattern : winPatterns) { if(board[pattern[0]] != ' ' && board[pattern[0]] == board[pattern[1]] && board[pattern[1]] == board[pattern[2]]) { return true; } } return false; } private static boolean isBoardFull() { for(char c : board) { if(c == ' ') return false; } return true; } }

Фрактал Мандельброта на Python (текстовый вывод)

Python

Вывод фрактала как ASCII-арт — работает в любом Python-компиляторе без matplotlib.

Показать код
def mandelbrot(h, w, max_iter): for j in range(h): row = "" for i in range(w): c = complex(-2 + 2.5 * i / w, -1.25 + 2.5 * j / h) z = 0 n = 0 while abs(z) <= 2 and n < max_iter: z = z*z + c n += 1 if n == max_iter: row += " " # вне множества else: row += "#" # внутри множества print(row) mandelbrot(30, 80, 30)

Как запустить: Скопируйте код → зайдите на Online Python → вставьте → нажмите "Run".
Увидите ASCII-фрактал в консоли — как в старых терминалах!

Калькулятор на Python (консольный)

Python

Базовый калькулятор: +, -, *, /, %, sqrt — работает в консоли без GUI.

Показать код
import math def calculator(): print("Калькулятор (введите 'quit' для выхода)") while True: expr = input("Введите выражение: ") if expr.lower() == 'quit': break try: expr = expr.replace('sqrt', 'math.sqrt') result = eval(expr) print("Результат:", result) except: print("Ошибка!") calculator()
